```cpp
#include <Arduino.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "freertos/semphr.h"

// Mutex and Semaphore
SemaphoreHandle_t resourceMutex;
SemaphoreHandle_t lowTaskReady;


// =====================================
// LOW PRIORITY TASK
// =====================================

void LowPriorityTask(void *pvParameters)
{
    Serial.println("\nLow Priority Task Started");

    // Acquire shared resource
    xSemaphoreTake(resourceMutex, portMAX_DELAY);

    Serial.println("Low Task: Mutex Acquired");
    Serial.println("Low Task: Holding Shared Resource");

    // Inform High Priority Task
    xSemaphoreGive(lowTaskReady);

    // Simulate resource processing
    unsigned long startTime = millis();

    while (millis() - startTime < 2000)
    {
        // Busy processing
    }

    Serial.print("Low Task: Current Priority = ");
    Serial.println(uxTaskPriorityGet(NULL));

    Serial.println("Low Task: Releasing Mutex");

    // Release mutex
    xSemaphoreGive(resourceMutex);

    Serial.print("Low Task: Restored Priority = ");
    Serial.println(uxTaskPriorityGet(NULL));

    Serial.println("Low Task: Resource Released");

    vTaskDelete(NULL);
}


// =====================================
// HIGH PRIORITY TASK
// =====================================

void HighPriorityTask(void *pvParameters)
{
    // Wait until Low Task acquires mutex
    xSemaphoreTake(lowTaskReady, portMAX_DELAY);

    Serial.println("\nHigh Priority Task Started");
    Serial.println("High Task: Requesting Shared Resource");

    // Request shared resource
    xSemaphoreTake(resourceMutex, portMAX_DELAY);

    Serial.println("High Task: Mutex Acquired");
    Serial.println("High Task: Accessing Shared Resource");

    vTaskDelay(pdMS_TO_TICKS(500));

    // Release mutex
    xSemaphoreGive(resourceMutex);

    Serial.println("High Task: Mutex Released");

    vTaskDelete(NULL);
}


// =====================================
// MEDIUM PRIORITY TASK
// =====================================

void MediumPriorityTask(void *pvParameters)
{
    // Wait for High Task to block on mutex
    vTaskDelay(pdMS_TO_TICKS(400));

    Serial.println("\nMedium Priority Task Started");
    Serial.println("Medium Task: Background Processing");

    unsigned long startTime = millis();

    while (millis() - startTime < 1000)
    {
        // CPU-intensive processing
    }

    Serial.println("Medium Task: Processing Completed");

    vTaskDelete(NULL);
}


// =====================================
// SETUP FUNCTION
// =====================================

void setup()
{
    Serial.begin(115200);

    delay(1000);

    Serial.println("\n================================");
    Serial.println("Priority Inversion Demonstration");
    Serial.println("================================");

    // Create Mutex
    resourceMutex = xSemaphoreCreateMutex();

    // Create Binary Semaphore
    lowTaskReady = xSemaphoreCreateBinary();

    if (resourceMutex == NULL || lowTaskReady == NULL)
    {
        Serial.println("Creation Failed!");
        return;
    }

    Serial.println("Mutex Created Successfully");

    // Create Low Priority Task
    xTaskCreatePinnedToCore(
        LowPriorityTask,
        "Low Priority",
        4096,
        NULL,
        1,
        NULL,
        1
    );

    // Create High Priority Task
    xTaskCreatePinnedToCore(
        HighPriorityTask,
        "High Priority",
        4096,
        NULL,
        3,
        NULL,
        1
    );

    // Create Medium Priority Task
    xTaskCreatePinnedToCore(
        MediumPriorityTask,
        "Medium Priority",
        4096,
        NULL,
        2,
        NULL,
        1
    );
}


// =====================================
// LOOP FUNCTION
// =====================================

void loop()
{
    vTaskDelay(pdMS_TO_TICKS(1000));
}
```
